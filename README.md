# lp-git-studi

- Зміна
	Команди для консолі Git Bash 

- git --version 		// Перевірка яка зараз версія Git встановлена
- clear 			// Почистити консоль 
- git clone " Силка з Git Hub " // Клонування репозиторію створеного в ( Git Hub )
- git status                    // Перевірка  які файли змінились 
- git add ( І назва файлу )	// Виділити конкретний файл
- git add . 			// Виділити всі файли які змінювались

	Навігація 

- pwd      // Показує поточний каталог 
- ls       // Показує вайлй і вибраній папці ( Крім заховани )
- ls -f    // Показує вайлй і вибраній папці ( І заховані файли теж показує )
- cd c:/   // Перейти в конкретний каталог 
- cd -     // Повернутись назад 
- cd ..    // Вийти на один рівень в верх 
- cd ../.. // Вийти на два рівені в верх 

	Створення каталогів

- mkdir  	        // Створення папки 
- cd !$ 	        // Перейти в тільки що створену папку 
- mkdir -p {app1,app2}  // Створити зразу декілька папок 
- mkdir -p app/{css,js} // Створення зразу декілька вкладених папок 

	Створення файлів

- touch index.html 				 // Створення файлу типу index.html
- touch app/{css/main.css,js/main.js,index.html} // Створення відразу декілька файлів ( Команда прописується без пробілів ) 

	Видалення Файлів

- touch 	// Дозволяє створити файли 
- rm test 	// Видалити пусту папку test
- rm -r test 	// Видалити папку test з її вмістом 

	Переміщення файлів 

- mv app1/*.* app2 // Перемістити всі файли з папки app1 в папку app2
