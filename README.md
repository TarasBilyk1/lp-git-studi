	1 КРОК Для збереження і загрузки репозиторію 

- git --version 		// Перевірка яка зараз версія Git встановлена
- git clone " Силка з Git Hub " // Клонування репозиторію створеного в ( Git Hub )
- ls       			// Показує файлй і вибраній папці ( Крім заховани )
- cd lp-git-studi/		// Вибір репозиторію

- git status                    // Перевірка  як змінились файли  
- git add . 			// Виділити всі файли які змінювались тепер файли очікують на те що ми будимо з ними робити 
- git status                    // Перевірка  як змінились файли 

-----------------------------------------------------------------------------------------------------------------
	2 КРОК Збереження в локальний репозиторій 

- git commit -m " add info "	// Комітимо зміни ( Зафіксовуємо зміи і додаємо коментар до наших змін (-m) )
- git status                    // Перевірка  як змінились файли  
- git push 			// Загружаємо зміни ( Файли відправляємо в GitHub )
- git status                    // Перевірка  як змінились файли 

-----------------------------------------------------------------------------------------------------------------
	3 Крок щоб зберегти чужі коміти 

- git pull 			// Збереження коміту ( Змін іншими робітниками і тд ) з GitHub
- git status                    // Перевірка  як змінились файли 

-----------------------------------------------------------------------------------------------------------------
	Кроки для створення нових Branch 
- git checkout -b future_1 	// Створення нового Branch до проекту 
- git status                    // Перевірка  як змінились файли ( Вписуємо для перевірки в якій ми гілці )
	Для загрузки файлів в новий Branch 
- git add . 			// Виділити всі файли які змінювались тепер файли очікують на те що ми будимо з ними робити 
- git commit -m " New branch "	// Комітимо зміни ( Зафіксовуємо зміи і додаємо коментар до наших змін (-m) )
- git push 			// Загружаємо зміни ( Файли відправляємо в GitHub )

І для нового Branch копіюємо функцію надану Git 

- git push --set-upstream origin feature_1 

- git branch           		// Переглянути існуючі гілки або створити нову

- git branch -b 		// Створити нову гілку

- git branch -a 		// Переглянути існуючі гілки локально і на сервері

-----------------------------------------------------------------------------------------------------------------
	Повернення з Branch в Main гілку ( За мержити )
- git checkout main	// Повернення в гілку Main
- git merge future_1 	// Втягування з future_1 в Main зміни 
- git push		// Загружаємо зміни ( Файли відправляємо в GitHub )

	Для переходу між бренчами
 
- git checkout ( Назва Branch )		// Зміна між гілками

	Якщо зробив роботу не в тому бранчі і потрібно перенести або не тоді почав роботу

- git status 
- git checkout main	// Не вийде зайти в мейн бо данні не збережені зміни 
- git stash 		// Заносимо роботу яку зробили в бранчі в стеш і можемо  
- git checkout main	// Тепер зможемо бо ми зберегли данні в stash

-----------------------------------------------------------------------------------------------------------------
	
	Видалення branch

- git branch --delete <branch_name>		// Локально видалити
or
- git branch --delete --force <branch_name>  	// Локально видалити

- git push origin --delete <branch_name>  	// Видалити з серверу github

-----------------------------------------------------------------------------------------------------------------

Функції зберігання і дропу stash 

- git stash pop 	// І тепер данні зявляться зі stash в різних бренчах

- git list		// Для перегляду які є ( stash )

- git stash [save]["my_stash info"] // Зберігаємо зміни і придумуємо комент можемо зберігати таких змін ( stash )  
				    // декілька і вони будуть збережені як [stash@{0-1-2-2-3...}]
- git stash pop   [stash@{0}]       // Дляповернення конкретного ( stash )
- git stash apply [stash@{0}]       // Дляповернення конкретного ( stash )

-----------------------------------------------------------------------------------------------------------------
	Можливі проблеми

Якщо вибиває помилку ( Author identity unknown *** Please tell me who you are.)
то потрібно вставити ці команди 

- git config --global user.email "hinaataras@gmail.com" //
- git config --global user.name "TarasBilyk1"		// Встановлення глобально для всіх репозиторіїв які 
будуть працювати з цього пк що це за людина комітить зміни ( Почта і імя акаунту GitHub )

Якщо потрібно щоб почта і імя використовувалось лише для цього репозиторію то потрібно з команд забрати 
( --global )
 

! Якщо ти змінив щось в файлі на пк то зроби ( commit push ) і тоді лише зберігай чужі зміни ( Pull )


Мерж це злиття двох комітів і тому бажано робитияк умога менше комітів в мейні і гілках одночасно бо може 
вилізти багато помилок !!!
